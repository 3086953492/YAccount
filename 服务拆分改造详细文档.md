# YAccount 服务拆分改造详细文档

## 🎯 改造目标

将当前混合的账号认证系统拆分为两个独立的微服务：

1. **认证服务 (Auth Service)** - 专门处理OAuth 2.0认证授权
2. **账号服务 (Account Service)** - 专门处理用户账号管理

## 📋 当前架构问题分析

### 问题概述
- 认证逻辑和业务逻辑耦合严重
- 职责不清晰，难以独立扩展
- 不符合微服务单一职责原则
- 违背OAuth 2.0标准架构设计

### 现状分析
```
当前单体服务 YAccount
├── 认证功能 (应该独立)
│   ├── OAuth客户端管理
│   ├── 授权码管理  
│   ├── Token管理
│   └── 权限验证
├── 账号功能 (保留)
│   ├── 用户管理
│   ├── 用户信息
│   └── 用户权限
└── 系统功能 (保留)
    └── 系统信息管理
```

### 服务职责划分

#### 🔐 认证服务 (Auth Service)
**端口**: 8080  
**职责**: OAuth 2.0 授权服务器

**核心功能**:
- OAuth 2.0 标准端点实现
- 客户端注册和管理
- 授权码生成和验证
- 访问令牌颁发和管理
- 刷新令牌处理
- Token 内省和撤销
- 用户登录验证（调用账号服务）

**API 端点**:
```
POST   /oauth/authorize       # 授权端点
POST   /oauth/token          # 令牌端点  
POST   /oauth/introspect     # 令牌内省
POST   /oauth/revoke         # 令牌撤销
GET    /oauth/clients        # 客户端列表
POST   /oauth/clients        # 注册客户端
PUT    /oauth/clients/:id    # 更新客户端
DELETE /oauth/clients/:id    # 删除客户端
GET    /.well-known/oauth-authorization-server  # 服务发现
```

#### 👥 账号服务 (Account Service)  
**端口**: 8081  
**职责**: 用户资源服务器

**核心功能**:
- 用户注册和信息管理
- 用户权限和角色管理
- 个人资料维护
- 系统信息管理
- 受保护资源访问

**API 端点**:
```
POST   /api/v1/users/register    # 用户注册
GET    /api/v1/users             # 用户列表
GET    /api/v1/users/:id         # 用户详情
PUT    /api/v1/users/:id         # 更新用户
DELETE /api/v1/users/:id         # 删除用户
GET    /api/v1/users/profile     # 用户资料
PUT    /api/v1/users/profile     # 更新资料
GET    /api/v1/system/info       # 系统信息
PUT    /api/v1/system/info       # 更新系统信息
```

package services

import (
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
    "time"
    "auth-service/configs"
)

type UserLoginRequest struct {
    Username string `json:"username"`
    Password string `json:"password"`
}

type UserResponse struct {
    ID       uint   `json:"id"`
    Username string `json:"username"`
    Role     string `json:"role"`
    Nickname string `json:"nickname"`
}

// 调用账号服务验证用户登录
func ValidateUserLogin(username, password string) (*UserResponse, error) {
    config := configs.GetConfig()
    url := config.AccountService.BaseURL + "/internal/users/validate"
    
    req := UserLoginRequest{
        Username: username,
        Password: password,
    }
    
    jsonData, err := json.Marshal(req)
    if err != nil {
        return nil, err
    }
    
    client := &http.Client{
        Timeout: 30 * time.Second,
    }
    
    resp, err := client.Post(url, "application/json", bytes.NewBuffer(jsonData))
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("用户验证失败")
    }
    
    var result struct {
        Code int          `json:"code"`
        Data UserResponse `json:"data"`
    }
    
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }
    
    if result.Code != 0 {
        return nil, fmt.Errorf("用户验证失败")
    }
    
    return &result.Data, nil
}

// 获取用户信息（内部调用）
func GetUserInfo(userID uint) (*UserResponse, error) {
    config := configs.GetConfig()
    url := fmt.Sprintf("%s/internal/users/%d", config.AccountService.BaseURL, userID)
    
    client := &http.Client{
        Timeout: 30 * time.Second,
    }
    
    resp, err := client.Get(url)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    if resp.StatusCode != http.StatusOK {
        return nil, fmt.Errorf("获取用户信息失败")
    }
    
    var result struct {
        Code int          `json:"code"`
        Data UserResponse `json:"data"`
    }
    
    if err := json.NewDecoder(resp.Body).Decode(&result); err != nil {
        return nil, err
    }
    
    return &result.Data, nil
}

### 3.3 创建OAuth Token验证中间件

**文件**: `services/account-service/middleware/auth/oauth.go`
```go
package auth

import (
    "account-service/configs"
    "bytes"
    "encoding/json"
    "fmt"
    "net/http"
    "strings"
    "time"
    
    "github.com/gin-gonic/gin"
)

type TokenIntrospectResponse struct {
    Active    bool     `json:"active"`
    Sub       string   `json:"sub"`
    ClientID  string   `json:"client_id"`
    Scope     string   `json:"scope"`
    Exp       int64    `json:"exp"`
    TokenType string   `json:"token_type"`
}

// OAuth Token验证中间件
func OAuthMiddleware(requiredScopes ...string) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 提取Token
        authHeader := c.GetHeader("Authorization")
        if authHeader == "" {
            c.JSON(http.StatusUnauthorized, gin.H{"error": "未提供访问令牌"})
            c.Abort()
            return
        }

        tokenParts := strings.Split(authHeader, " ")
        if len(tokenParts) != 2 || strings.ToLower(tokenParts[0]) != "bearer" {
            c.JSON(http.StatusUnauthorized, gin.H{"error": "令牌格式错误"})
            c.Abort()
            return
        }

        token := tokenParts[1]

        // 调用认证服务进行Token内省
        introspectResp, err := introspectToken(token)
        if err != nil {
            c.JSON(http.StatusUnauthorized, gin.H{"error": "令牌验证失败"})
            c.Abort()
            return
        }

        if !introspectResp.Active {
            c.JSON(http.StatusUnauthorized, gin.H{"error": "令牌无效或已过期"})
            c.Abort()
            return
        }

        // 检查权限范围
        if len(requiredScopes) > 0 {
            tokenScopes := strings.Fields(introspectResp.Scope)
            if !hasRequiredScopes(tokenScopes, requiredScopes) {
                c.JSON(http.StatusForbidden, gin.H{"error": "权限不足"})
                c.Abort()
                return
            }
        }

        // 设置用户上下文
        c.Set("user_id", introspectResp.Sub)
        c.Set("client_id", introspectResp.ClientID)
        c.Set("scopes", strings.Fields(introspectResp.Scope))

        c.Next()
    }
}

// 调用认证服务进行Token内省
func introspectToken(token string) (*TokenIntrospectResponse, error) {
    config := configs.GetConfig()
    url := config.AuthService.BaseURL + config.AuthService.IntrospectEndpoint

    data := map[string]string{
        "token": token,
    }

    jsonData, err := json.Marshal(data)
    if err != nil {
        return nil, err
    }

    client := &http.Client{
        Timeout: 10 * time.Second,
    }

    resp, err := client.Post(url, "application/json", bytes.NewBuffer(jsonData))
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var introspectResp TokenIntrospectResponse
    if err := json.NewDecoder(resp.Body).Decode(&introspectResp); err != nil {
        return nil, err
    }

    return &introspectResp, nil
}

// 检查是否有所需的权限范围
func hasRequiredScopes(tokenScopes, requiredScopes []string) bool {
    for _, required := range requiredScopes {
        found := false
        for _, scope := range tokenScopes {
            if scope == required {
                found = true
                break
            }
        }
        if !found {
            return false
        }
    }
    return true
}
```

package controllers

import (
    "account-service/models"
    "account-service/services"
    "net/http"
    "strconv"

    "github.com/gin-gonic/gin"
)

// 内部用户验证接口（供认证服务调用）
func InternalValidateUser(c *gin.Context) {
    var req struct {
        Username string `json:"username" binding:"required"`
        Password string `json:"password" binding:"required"`
    }

    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{
            "code":    1,
            "message": "请求参数错误",
        })
        return
    }

    user, err := services.ValidateUserCredentials(req.Username, req.Password)
    if err != nil {
        c.JSON(http.StatusUnauthorized, gin.H{
            "code":    1,
            "message": "用户名或密码错误",
        })
        return
    }

    c.JSON(http.StatusOK, gin.H{
        "code": 0,
        "data": gin.H{
            "id":       user.ID,
            "username": user.Username,
            "role":     user.Role,
            "nickname": user.Nickname,
        },
    })
}

// 内部获取用户信息接口（供认证服务调用）
func InternalGetUser(c *gin.Context) {
    userIDStr := c.Param("id")
    userID, err := strconv.ParseUint(userIDStr, 10, 32)
    if err != nil {
        c.JSON(http.StatusBadRequest, gin.H{
            "code":    1,
            "message": "用户ID格式错误",
        })
        return
    }

    user, err := services.GetUserByID(uint(userID))
    if err != nil {
        c.JSON(http.StatusNotFound, gin.H{
            "code":    1,
            "message": "用户不存在",
        })
        return
    }

    c.JSON(http.StatusOK, gin.H{
        "code": 0,
        "data": gin.H{
            "id":       user.ID,
            "username": user.Username,
            "role":     user.Role,
            "nickname": user.Nickname,
        },
    })
}

    }

### 5.2 更新认证API

**文件**: `web/src/api/auth.ts`
```typescript
import { request } from '@/utils/request';
import { getApiBaseUrl } from '@/config/api';

const authRequest = request.create({
  baseURL: getApiBaseUrl('auth'),
});

// OAuth登录流程
export const initiateOAuthLogin = async (clientId: string, redirectUri: string) => {
  const params = new URLSearchParams({
    response_type: 'code',
    client_id: clientId,
    redirect_uri: redirectUri,
    scope: 'read write',
    state: generateRandomState(),
  });
  
  return `/oauth/authorize?${params.toString()}`;
};

// 使用授权码获取Token
export const exchangeCodeForToken = async (code: string, clientId: string, clientSecret: string, redirectUri: string) => {
  const response = await authRequest.post('/oauth/token', {
    grant_type: 'authorization_code',
    client_id: clientId,
    client_secret: clientSecret,
    code,
    redirect_uri: redirectUri,
  });
  
  return response.data;
};

// 刷新Token
export const refreshAccessToken = async (refreshToken: string, clientId: string, clientSecret: string) => {
  const response = await authRequest.post('/oauth/token', {
    grant_type: 'refresh_token',
    refresh_token: refreshToken,
    client_id: clientId,
    client_secret: clientSecret,
  });
  
  return response.data;
};

function generateRandomState(): string {
  return Math.random().toString(36).substring(2, 15) + 
         Math.random().toString(36).substring(2, 15);
}
```
